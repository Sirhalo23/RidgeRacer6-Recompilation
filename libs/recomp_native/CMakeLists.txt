cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(recomp_native LANGUAGES CXX)
    option(RECOMP_NATIVE_BUILD_TESTS "Build shared native wait tests" ON)
endif()
add_library(recomp_native STATIC src/wait_set.cpp)
set_target_properties(recomp_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(recomp::native ALIAS recomp_native)
target_include_directories(recomp_native PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(recomp_native PUBLIC cxx_std_20)
find_package(Threads REQUIRED)
target_link_libraries(recomp_native PUBLIC Threads::Threads)
if(RECOMP_NATIVE_BUILD_TESTS)
    enable_testing()
    add_executable(recomp_native_wait_set_tests tests/wait_set_tests.cpp)
    target_link_libraries(recomp_native_wait_set_tests PRIVATE recomp::native)
    add_test(NAME recomp_native_wait_set COMMAND recomp_native_wait_set_tests)
    set_tests_properties(recomp_native_wait_set PROPERTIES TIMEOUT 30)
endif()
